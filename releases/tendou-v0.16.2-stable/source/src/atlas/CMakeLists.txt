set(TARGET llama-atlas-engine)
add_executable(${TARGET} atlas-engine.cpp atlas-engine.h)
install(TARGETS ${TARGET} RUNTIME)
target_link_libraries(${TARGET} PRIVATE llama-common llama ggml ggml-base ggml-cpu ${CMAKE_THREAD_LIBS_INIT} ${CMAKE_DL_LIBS})
if (GGML_CUDA)
    target_link_libraries(${TARGET} PRIVATE ggml-cuda)
    target_compile_definitions(${TARGET} PRIVATE GGML_USE_CUDA)
endif()
if (WIN32)
    target_link_libraries(${TARGET} PRIVATE kernel32 user32 advapi32 ws2_32)
    target_compile_definitions(${TARGET} PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
target_compile_features(${TARGET} PRIVATE cxx_std_17)
